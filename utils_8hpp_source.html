22inline double wrap_angle_rad(double angle_rad, double min_val = -M_PI, double max_val = M_PI) {
23 double capped_angle_rad = angle_rad;
24 while (capped_angle_rad > max_val) capped_angle_rad -= 2 * M_PI;
25 while (capped_angle_rad < min_val) capped_angle_rad += 2 * M_PI;
26 return capped_angle_rad;
27}

36inline double computeLaneletYawAtArcLength(const lanelet::ConstLanelet& lanelet, double arc_length) {
37 const lanelet::ConstLineString2d centerline = lanelet.centerline2d();
38 const double lanelet_length = static_cast<double>(lanelet::geometry::length(centerline));
39 const double sample_distance = std::min(0.5, std::max(0.01, lanelet_length * 0.1));
40 const double before_arc_length = std::max(0.0, arc_length - sample_distance);
41 const double after_arc_length = std::min(lanelet_length, arc_length + sample_distance);
42 const lanelet::BasicPoint2d before_point = lanelet::geometry::interpolatedPointAtDistance(centerline, before_arc_length);
43 const lanelet::BasicPoint2d after_point = lanelet::geometry::interpolatedPointAtDistance(centerline, after_arc_length);
44 return std::atan2(after_point.y() - before_point.y(), after_point.x() - before_point.x());
45}
