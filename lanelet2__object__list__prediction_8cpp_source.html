44template <typename T>
45void Lanelet2ObjectListPrediction::declareAndLoadParameter(const std::string& name,
46 T& param,
47 const std::string& description,
48 const bool add_to_auto_reconfigurable_params,
49 const bool is_required,
50 const bool read_only,
51 const std::optional<double>& from_value,
52 const std::optional<double>& to_value,
53 const std::optional<double>& step_value,
54 const std::string& additional_constraints) {
55 rcl_interfaces::msg::ParameterDescriptor param_desc;
56 param_desc.description = description;
57 param_desc.additional_constraints = additional_constraints;
58 param_desc.read_only = read_only;
59
60 auto type = rclcpp::ParameterValue(param).get_type();
61
62 if (from_value.has_value() && to_value.has_value()) {
63 if constexpr (std::is_integral_v<T>) {
64 rcl_interfaces::msg::IntegerRange range;
65 range.set__from_value(static_cast<T>(from_value.value())).set__to_value(static_cast<T>(to_value.value()));
66 if (step_value.has_value()) range.set__step(static_cast<T>(step_value.value()));
67 param_desc.integer_range = {range};

69 rcl_interfaces::msg::FloatingPointRange range;
70 range.set__from_value(static_cast<T>(from_value.value())).set__to_value(static_cast<T>(to_value.value()));
71 if (step_value.has_value()) range.set__step(static_cast<T>(step_value.value()));
72 param_desc.floating_point_range = {range};
73 } else {
74 RCLCPP_WARN(this->get_logger(), "Parameter type of parameter '%s' does not support specifying a range", name.c_str());
75 }
76 }
77
78 this->declare_parameter(name, type, param_desc);
79
80 try {
81 param = this->get_parameter(name).get_value<T>();
82 std::stringstream ss;
83 ss << "Loaded parameter '" << name << "': ";
84 if constexpr (is_vector_v<T>) {
85 ss << "[";
86 for (const auto& element : param) ss << element << (&element != &param.back() ? ", " : "");
87 ss << "]";
88 } else {
89 ss << param;
90 }
91 RCLCPP_INFO_STREAM(this->get_logger(), ss.str());
92 } catch (rclcpp::exceptions::ParameterUninitializedException&) {
93 if (is_required) {
94 RCLCPP_FATAL_STREAM(this->get_logger(), "Missing required parameter '" << name << "', exiting");
95 exit(EXIT_FAILURE);
96 } else {
97 std::stringstream ss;
98 ss << "Missing parameter '" << name << "', using default value: ";
99 if constexpr (is_vector_v<T>) {
100 ss << "[";
101 for (const auto& element : param) ss << element << (&element != &param.back() ? ", " : "");
102 ss << "]";
103 } else {
104 ss << param;
105 }
106 RCLCPP_WARN_STREAM(this->get_logger(), ss.str());
107 this->set_parameters({rclcpp::Parameter(name, rclcpp::ParameterValue(param))});
108 }
109 }
110
111 if (add_to_auto_reconfigurable_params) {
112 std::function<void(const rclcpp::Parameter&)> setter = [&param](const rclcpp::Parameter& p) { param = p.get_value<T>(); };
113 auto_reconfigurable_params_.push_back(std::make_tuple(name, setter));
114 }

135void Lanelet2ObjectListPrediction::setup() {
136 // TF listener for transforming incoming object lists into the map frame
137 tf_buffer_ = std::make_unique<tf2_ros::Buffer>(this->get_clock());
138 tf_listener_ = std::make_shared<tf2_ros::TransformListener>(*tf_buffer_);
139
140 // map interface
141 ll2_interface_ = std::make_unique<Lanelet2MapInterface>(*this, ll2_map_server_name_);
142
143 // callback for dynamic parameter configuration
144 parameters_callback_ = this->add_on_set_parameters_callback(
145 std::bind(&Lanelet2ObjectListPrediction::parametersCallback, this, std::placeholders::_1));
146
147 // subscriber for handling incoming messages
148 subscriber_ = this->create_subscription<perception_msgs::msg::ObjectList>(
149 "~/tracked_object_list", 1, std::bind(&Lanelet2ObjectListPrediction::objectListCallback, this, std::placeholders::_1));
150 RCLCPP_INFO(this->get_logger(), "Subscribed to '%s'", subscriber_->get_topic_name());
151
152 // publisher for publishing outgoing messages
153 publisher_ = this->create_publisher<perception_msgs::msg::ObjectList>("~/object_list", 1);
154 RCLCPP_INFO(this->get_logger(), "Publishing to '%s'", publisher_->get_topic_name());
155}

157void Lanelet2ObjectListPrediction::objectListCallback(const perception_msgs::msg::ObjectList::ConstSharedPtr& msg) {
158 RCLCPP_DEBUG(this->get_logger(), "Message received with stamp: '%d'", msg->header.stamp.sec);
159
160 if (!checkMap(true)) {
161 RCLCPP_WARN(this->get_logger(), "Lanelet2 map is not loaded yet, skipping object list");
162 return;
163 }
164
165 perception_msgs::msg::ObjectList object_list_map_frame;
166 if (msg->header.frame_id != ll2_interface_->map_frame_id_) {
167 try {
168 object_list_map_frame = tf_buffer_->transform(*msg, ll2_interface_->map_frame_id_, tf2::durationFromSec(0.1));
169 } catch (tf2::TransformException& ex) {
170 RCLCPP_ERROR(this->get_logger(), "Could not transform object list from frame '%s' to frame '%s': %s. Skipping object list.",
171 msg->header.frame_id.c_str(), ll2_interface_->map_frame_id_.c_str(), ex.what());
172 return;
173 }
174 } else {
175 object_list_map_frame = *msg;
176 }
177
178 std::vector<PredictionObject> prediction_objects = matchObjectListToMap(object_list_map_frame);
179 std::size_t matched_object_count = 0;
180 for (PredictionObject& prediction_object : prediction_objects) {
181 if (!prediction_object.lanelet_matches.empty()) {
182 ++matched_object_count;
183 }
184 prediction_object.object.state_predictions =
185 createPredictionsForMatchedObject(prediction_object, object_list_map_frame.header.stamp);
186 }
187 RCLCPP_DEBUG(this->get_logger(), "Matched %zu/%zu objects to at least one lanelet", matched_object_count,
188 object_list_map_frame.objects.size());
189
190 perception_msgs::msg::ObjectList out_msg = object_list_map_frame;
191 out_msg.objects.clear();
192 out_msg.objects.reserve(prediction_objects.size());
193 for (const PredictionObject& prediction_object : prediction_objects) {
194 out_msg.objects.push_back(prediction_object.object);
195 }
196
197 publisher_->publish(out_msg);
198 RCLCPP_DEBUG(this->get_logger(), "Message published with stamp: '%d'", out_msg.header.stamp.sec);
199}

201std::vector<Lanelet2ObjectListPrediction::PredictionObject> Lanelet2ObjectListPrediction::matchObjectListToMap(
202 const perception_msgs::msg::ObjectList& object_list) const {
203 std::vector<PredictionObject> prediction_objects;
204 prediction_objects.reserve(object_list.objects.size());
205
206 const auto map = ll2_interface_->getMapPtr();
207 if (map == nullptr) {
208 RCLCPP_ERROR(this->get_logger(), "Lanelet2 map pointer is null, cannot match objects to lanelets");
209 return prediction_objects;
210 }
211
212 for (std::size_t object_index = 0; object_index < object_list.objects.size(); ++object_index) {
213 PredictionObject prediction_object;
214 prediction_object.object = object_list.objects[object_index];
215
216 const auto classification = perception_msgs::object_access::getClassWithHighestProbability(prediction_object.object);
217 // No lanelet matching for pedestrians, as they are not constrained to the road network
218 if (classification.type == perception_msgs::msg::ObjectClassification::PEDESTRIAN) {
219 prediction_objects.push_back(prediction_object);
220 continue;
221 }
222
223 geometry_msgs::msg::Point position;
224 double object_yaw = 0.0;
225 try {
226 position = perception_msgs::object_access::getPosition(prediction_object.object);
227 object_yaw = perception_msgs::object_access::getYaw(prediction_object.object);
228 } catch (const std::exception& ex) {
229 RCLCPP_WARN(this->get_logger(), "Could not read position or yaw of object %zu: %s", object_index, ex.what());
230 prediction_objects.push_back(prediction_object);
231 continue;
232 }
233
234 const lanelet::BasicPoint2d position_2d(position.x, position.y);
235 const auto candidate_lanelets =
236 lanelet::geometry::findWithin2d(map->laneletLayer, position_2d, lanelet_match_max_distance_m_);
237
238 for (const auto& candidate_lanelet : candidate_lanelets) {
239 lanelet::ConstLanelet lanelet = candidate_lanelet.second;
240 lanelet::ConstLanelet matched_lanelet = lanelet;
241 double start_arc_length = lanelet::geometry::toArcCoordinates(lanelet.centerline2d(), position_2d).length;
242 const double lanelet_length = static_cast<double>(lanelet::geometry::length(lanelet.centerline2d()));
243 start_arc_length = std::clamp(start_arc_length, 0.0, lanelet_length);
244 double orientation_difference = 0.0;
245
246 const double lanelet_yaw = computeLaneletYawAtArcLength(lanelet, start_arc_length);
247 const double inverted_arc_length = lanelet_length - start_arc_length;
248 const double inverted_yaw = computeLaneletYawAtArcLength(lanelet.invert(), inverted_arc_length);
249 const double lanelet_difference = std::abs(wrap_angle_rad(object_yaw - lanelet_yaw));
250 const double inverted_difference = std::abs(wrap_angle_rad(object_yaw - inverted_yaw));
251
252 if (inverted_difference < lanelet_difference) {
253 matched_lanelet = lanelet.invert();
254 start_arc_length = inverted_arc_length;
255 orientation_difference = inverted_difference;
256 } else {
257 orientation_difference = lanelet_difference;
258 }
259
260 if (orientation_difference > lanelet_match_max_yaw_diff_rad_) {
261 continue;
262 }
263
264 prediction_object.lanelet_matches.push_back(
265 LaneletMatch{matched_lanelet, candidate_lanelet.first, start_arc_length, orientation_difference});
266 }
267
268 if (prediction_object.lanelet_matches.empty()) {
269 RCLCPP_DEBUG(this->get_logger(), "Object %zu did not match any lanelet within %.2f m", object_index,
270 lanelet_match_max_distance_m_);
271 } else {
272 RCLCPP_DEBUG(this->get_logger(), "Object %zu matched to %zu lanelet candidate(s)", object_index,
273 prediction_object.lanelet_matches.size());
274 }
275 prediction_objects.push_back(prediction_object);
276 }
277
278 return prediction_objects;
279}

281std::vector<perception_msgs::msg::ObjectStatePrediction> Lanelet2ObjectListPrediction::createPredictionsForMatchedObject(
282 const PredictionObject& prediction_object, const builtin_interfaces::msg::Time& base_time) const {
283 std::vector<perception_msgs::msg::ObjectStatePrediction> predictions;
284 if (!prediction_object.lanelet_matches.empty()) {
285 predictions = createMapBasedPredictions(prediction_object, base_time);
286 }
287
288 if (predictions.empty()) {
289 if (unmatched_object_prediction_mode_ == "static") {
290 predictions.push_back(createStationaryPrediction(prediction_object.object, base_time));
291 } else {
292 predictions.push_back(createConstantVelocityPrediction(prediction_object.object, base_time));
293 }
294 }
295
296 const double probability = 1.0 / static_cast<double>(predictions.size());
297 for (perception_msgs::msg::ObjectStatePrediction& prediction : predictions) {
298 prediction.probability = probability;
299 }
300 return predictions;
301}

303std::vector<perception_msgs::msg::ObjectStatePrediction> Lanelet2ObjectListPrediction::createMapBasedPredictions(
304 const PredictionObject& prediction_object, const builtin_interfaces::msg::Time& base_time) const {
305 std::vector<perception_msgs::msg::ObjectStatePrediction> predictions;
306 if (routing_graph_ == nullptr) {
307 RCLCPP_WARN(this->get_logger(), "Routing graph is not available, cannot create lanelet predictions");
308 return predictions;
309 }
310
311 double speed = 0.0;
312
313 try {
314 speed = perception_msgs::object_access::getVelocityMagnitude(prediction_object.object);
315 } catch (const std::exception& ex) {
316 RCLCPP_WARN(this->get_logger(), "Could not read velocity for lanelet prediction: %s", ex.what());
317 return predictions;
318 }
319
320 const std::size_t sample_count = getPredictionSampleCount();
321 const double max_travel_distance = speed * prediction_sample_interval_s_ * static_cast<double>(sample_count);
322
323 for (const LaneletMatch& match : prediction_object.lanelet_matches) {
324 lanelet::routing::LaneletPaths routes;
325 if (max_travel_distance <= std::numeric_limits<double>::epsilon()) {
326 routes.push_back(lanelet::routing::LaneletPath({match.lanelet}));
327 } else {
328 lanelet::routing::PossiblePathsParams params;
329 params.routingCostLimit = max_travel_distance + match.start_arc_length;
330 params.includeShorterPaths = true;
331 params.includeLaneChanges = false;
332 try {
333 routes = routing_graph_->possiblePaths(match.lanelet, params);
334 } catch (const std::exception& ex) {
335 RCLCPP_WARN(this->get_logger(), "Could not create lanelet routes from matched lanelet: %s", ex.what());
336 continue;
337 }
338 if (routes.empty()) {
339 routes.push_back(lanelet::routing::LaneletPath({match.lanelet}));
340 }
341 }
342
343 for (const lanelet::routing::LaneletPath& lanelet_route : routes) {
344 perception_msgs::msg::ObjectStatePrediction prediction;
345 prediction.states.reserve(sample_count);
346 for (std::size_t sample_index = 0; sample_index < sample_count; ++sample_index) {
347 const double travel_distance = speed * prediction_sample_interval_s_ * static_cast<double>(sample_index + 1);
348 prediction.states.push_back(sampleStateOnLaneletRoute(prediction_object.object.state, lanelet_route,
349 match.start_arc_length, travel_distance, base_time, sample_index));
350 }
351 predictions.push_back(prediction);
352 }
353 }
354
355 return predictions;
356}

358perception_msgs::msg::ObjectStatePrediction Lanelet2ObjectListPrediction::createStationaryPrediction(
359 const perception_msgs::msg::Object& object, const builtin_interfaces::msg::Time& base_time) const {
360 perception_msgs::msg::ObjectStatePrediction prediction;
361 const std::size_t sample_count = getPredictionSampleCount();
362 prediction.states.reserve(sample_count);
363
364 geometry_msgs::msg::Point position = perception_msgs::object_access::getPosition(object);
365 const double yaw = perception_msgs::object_access::getYaw(object);
366 geometry_msgs::msg::Vector3 velocity;
367 velocity.x = 0.0;
368 velocity.y = 0.0;
369 velocity.z = 0.0;
370
371 for (std::size_t sample_index = 0; sample_index < sample_count; ++sample_index) {
372 perception_msgs::msg::ObjectState state = object.state;
373 setPredictedStateKinematics(state, position.x, position.y, position.z, yaw, velocity, base_time, sample_index);
374 prediction.states.push_back(state);
375 }
376 return prediction;
377}

379perception_msgs::msg::ObjectStatePrediction Lanelet2ObjectListPrediction::createConstantVelocityPrediction(
380 const perception_msgs::msg::Object& object, const builtin_interfaces::msg::Time& base_time) const {
381 perception_msgs::msg::ObjectStatePrediction prediction;
382 const std::size_t sample_count = getPredictionSampleCount();
383 prediction.states.reserve(sample_count);
384
385 geometry_msgs::msg::Point position;
386 geometry_msgs::msg::Vector3 velocity;
387 double yaw = 0.0;
388 try {
389 position = perception_msgs::object_access::getPosition(object);
390 velocity = perception_msgs::object_access::getVelocityXYZ(object);
391 yaw = perception_msgs::object_access::getYaw(object);
392 } catch (const std::exception& ex) {
393 RCLCPP_WARN(this->get_logger(), "Could not read velocity for kinematic prediction, using static fallback: %s", ex.what());
394 return createStationaryPrediction(object, base_time);
395 }
396
397 for (std::size_t sample_index = 0; sample_index < sample_count; ++sample_index) {
398 const double time_offset = prediction_sample_interval_s_ * static_cast<double>(sample_index + 1);
399 perception_msgs::msg::ObjectState state = object.state;
400 setPredictedStateKinematics(state, position.x + velocity.x * time_offset, position.y + velocity.y * time_offset,
401 position.z + velocity.z * time_offset, yaw, velocity, base_time, sample_index);
402 prediction.states.push_back(state);
403 }
404 return prediction;
405}

407perception_msgs::msg::ObjectState Lanelet2ObjectListPrediction::sampleStateOnLaneletRoute(
408 const perception_msgs::msg::ObjectState& base_state,
409 const lanelet::routing::LaneletPath& route,
410 double start_arc_length,
411 double travel_distance,
412 const builtin_interfaces::msg::Time& base_time,
413 std::size_t sample_index) const {
414 perception_msgs::msg::ObjectState state = base_state;
415 if (route.empty()) {
416 RCLCPP_WARN(this->get_logger(), "Lanelet route is empty, cannot sample lanelet prediction");
417 return state;
418 }
419
420 const double speed = travel_distance / (prediction_sample_interval_s_ * static_cast<double>(sample_index + 1));
421 double distance_on_route = start_arc_length + travel_distance;
422 geometry_msgs::msg::Point fallback_position;
423 try {
424 fallback_position = perception_msgs::object_access::getPosition(base_state);
425 } catch (const std::exception&) {
426 fallback_position.x = 0.0;
427 fallback_position.y = 0.0;
428 fallback_position.z = 0.0;
429 }
430
431 for (std::size_t route_index = 0; route_index < route.size(); ++route_index) {
432 const lanelet::ConstLineString2d centerline = route[route_index].centerline2d();
433 if (centerline.size() < 2) {
434 continue;
435 }
436
437 const double lanelet_length = static_cast<double>(lanelet::geometry::length(centerline));
438 const bool is_last_lanelet = route_index + 1 == route.size();
439 if (distance_on_route > lanelet_length && !is_last_lanelet) {
440 distance_on_route -= lanelet_length;
441 continue;
442 }
443
444 const double arc_length = std::clamp(distance_on_route, 0.0, lanelet_length);
445 const lanelet::BasicPoint2d point = lanelet::geometry::interpolatedPointAtDistance(centerline, arc_length);
446 const double yaw_sample_distance = std::min(0.5, std::max(0.01, lanelet_length * 0.1));
447 double before_arc_length = std::max(0.0, arc_length - yaw_sample_distance);
448 double after_arc_length = std::min(lanelet_length, arc_length + yaw_sample_distance);
449 if (after_arc_length <= before_arc_length) {
450 before_arc_length = 0.0;
451 after_arc_length = lanelet_length;
452 }
453
454 const lanelet::BasicPoint2d before_point = lanelet::geometry::interpolatedPointAtDistance(centerline, before_arc_length);
455 const lanelet::BasicPoint2d after_point = lanelet::geometry::interpolatedPointAtDistance(centerline, after_arc_length);
456 const double yaw = std::atan2(after_point.y() - before_point.y(), after_point.x() - before_point.x());
457 geometry_msgs::msg::Vector3 velocity;
458 velocity.x = speed * std::cos(yaw);
459 velocity.y = speed * std::sin(yaw);
460 velocity.z = 0.0;
461 setPredictedStateKinematics(state, point.x(), point.y(), fallback_position.z, yaw, velocity, base_time, sample_index);
462 return state;
463 }
464
465 const lanelet::ConstLineString2d last_centerline = route.back().centerline2d();
466 const double last_length = static_cast<double>(lanelet::geometry::length(last_centerline));
467 const lanelet::BasicPoint2d point = lanelet::geometry::interpolatedPointAtDistance(last_centerline, last_length);
468 geometry_msgs::msg::Vector3 velocity;
469 velocity.x = 0.0;
470 velocity.y = 0.0;
471 velocity.z = 0.0;
472 setPredictedStateKinematics(state, point.x(), point.y(), fallback_position.z, 0.0, velocity, base_time, sample_index);
473 return state;
474}

476std::size_t Lanelet2ObjectListPrediction::getPredictionSampleCount() const {
477 return std::max<std::size_t>(1, static_cast<std::size_t>(std::floor(prediction_horizon_s_ / prediction_sample_interval_s_)));
478}

480void Lanelet2ObjectListPrediction::setPredictedStateKinematics(perception_msgs::msg::ObjectState& state,
481 double x,
482 double y,
483 double z,
484 double yaw,
485 const geometry_msgs::msg::Vector3& velocity,
486 const builtin_interfaces::msg::Time& base_time,
487 std::size_t sample_index) const {
488 state.header.frame_id = ll2_interface_->map_frame_id_;
489 const rclcpp::Time stamp(base_time);
490 const double time_offset = prediction_sample_interval_s_ * static_cast<double>(sample_index + 1);
491 const rclcpp::Time future_stamp = stamp + rclcpp::Duration::from_seconds(time_offset);
492 const int64_t future_nanoseconds = future_stamp.nanoseconds();
493 state.header.stamp.sec = static_cast<int32_t>(future_nanoseconds / 1000000000);
494 state.header.stamp.nanosec = static_cast<uint32_t>(future_nanoseconds % 1000000000);
495
496 geometry_msgs::msg::Point position;
497 position.x = x;
498 position.y = y;
499 position.z = z;
500 perception_msgs::object_access::setPosition(state, position, false);
501
502 try {
503 perception_msgs::object_access::setVelocityXYZYaw(state, velocity, yaw, false);
504 } catch (const std::exception& ex) {
505 RCLCPP_DEBUG(this->get_logger(), "Could not set predicted velocity, setting yaw only: %s", ex.what());
506 perception_msgs::object_access::setYaw(state, yaw, false);
507 }
508}

510void Lanelet2ObjectListPrediction::rebuildRoutingGraphFromMap() {
511 routing_graph_.reset();
512 routing_graph_map_ = ll2_interface_->getMapPtr();
513 if (routing_graph_map_ == nullptr) {
514 RCLCPP_WARN(this->get_logger(), "Lanelet2 map pointer is null, cannot build routing graph");
515 return;
516 }
517
518 lanelet::traffic_rules::TrafficRulesUPtr traffic_rules = lanelet::traffic_rules::TrafficRulesFactory::create(

526 bool map_status = ll2_interface_->map_loaded_;
527 // update routing graph on map update
528 if (handle_update && ll2_interface_->update_pending_ && ll2_interface_->map_loaded_) {
529 ll2_interface_->update_pending_ = false;
530 map_status = map_status && !ll2_interface_->update_pending_;
531 }

534 }
