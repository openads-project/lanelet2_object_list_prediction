39// Paths against the legal direction, extending `path` of inverted lanelets along the legal predecessors.
40lanelet::routing::LaneletPaths oppositeDirectionPaths(const lanelet::ConstLanelets& path,
41 double remaining_cost,
42 const lanelet::routing::RoutingGraph& routing_graph) {
43 remaining_cost -= static_cast<double>(lanelet::geometry::length(path.back().centerline2d()));
44 lanelet::routing::LaneletPaths paths;
45 for (const lanelet::ConstLanelet& predecessor :
46 remaining_cost > 0.0 ? routing_graph.previous(path.back().invert(), false) : lanelet::ConstLanelets{}) {
47 lanelet::ConstLanelets extended = path;
48 extended.push_back(predecessor.invert());
49 const lanelet::routing::LaneletPaths extended_paths = oppositeDirectionPaths(extended, remaining_cost, routing_graph);
50 paths.insert(paths.end(), extended_paths.begin(), extended_paths.end());
51 }
52 if (paths.empty()) paths.emplace_back(path);
53 return paths;
54}

56// Chord length to smooth: increase in case of high centerline jitter, decrease for sharp turns
57constexpr double kHeadingChordLengthM = 3.0;
58// Removes centerline steps shorter than about kHeadingMedianHalfWidth - kHeadingChordLengthM = 3 m, keeping sustained turns
59constexpr int kHeadingMedianHalfWidth = 6;
60
61// Curvature every meter from `start_arc_length` along a line, from the change of its heading. The heading at
62// each meter is the median over ± kHeadingMedianHalfWidth m, as map centerlines have short sideways steps and kinks, which
63// would otherwise look like sharp curves. Turns of any length are kept, as they change the heading.
64std::vector<double> curvaturesAlongLine(const lanelet::CompoundLineString2d& line, double start_arc_length, std::size_t count) {
65 const double length = static_cast<double>(lanelet::geometry::length(line));
66 const int half_width = kHeadingMedianHalfWidth;
67
68 // 1. Raw heading every meter, as the direction of the chord from kHeadingChordLengthM / 2 before to
69 // kHeadingChordLengthM / 2 after that meter. They reach half_width + 1 m beyond the curvatures on both sides: 1 m
70 // because each curvature uses the median headings 1 m before and after it, and half_width m because each median
71 // uses the raw headings around it.
72 std::vector<double> headings;
73 for (int i = -1 - half_width; i <= static_cast<int>(count) + half_width; ++i) {
74 const double chord_start =
75 std::max(0.0, std::min(start_arc_length + i - kHeadingChordLengthM / 2.0, length - kHeadingChordLengthM));
76 const lanelet::BasicPoint2d chord = lanelet::geometry::interpolatedPointAtDistance(line, chord_start + kHeadingChordLengthM) -
77 lanelet::geometry::interpolatedPointAtDistance(line, chord_start);
78 const double heading = std::atan2(chord.y(), chord.x());
79 headings.push_back(headings.empty() ? heading : headings.back() + wrap_angle_rad(heading - headings.back()));
80 }
81
82 // 2. Median of the 2 * half_width + 1 raw headings around each meter, from 1 m before the first curvature to 1 m after
83 // the last. A short step or kink only changes a few headings in the window, which the median ignores. In a turn,
84 // the headings change steadily in one direction, so the median is the heading at the center and the turn is kept.
85 std::vector<double> median_headings;
86 for (auto window_start = headings.begin(); window_start + 2 * half_width < headings.end(); ++window_start) {
87 std::vector<double> window(window_start, window_start + 2 * half_width + 1);
88 std::nth_element(window.begin(), window.begin() + half_width, window.end());
89 median_headings.push_back(window[half_width]);
90 }
91
92 // 3. Curvature at each meter, as the change of the median heading per meter from 1 m before to 1 m after it
93 std::vector<double> curvatures(count);
94 for (std::size_t i = 0; i < count; ++i) curvatures[i] = std::abs(median_headings[i + 2] - median_headings[i]) / 2.0;
95 return curvatures;
96}

98constexpr int kSimulationStepsPerSample = 10; // steps short enough not to skip the speed limits set every meter

100// Travel distance and speed at each sample time along a route. Starting at `speed`, the object brakes with at most
101// `max_deceleration` early enough to pass curves with at most `max_lateral_acceleration`, to stop at `stop_distance`
102// and to stay `min_distance` behind its leader, and accelerates with at most `max_acceleration` back towards its initial
103// speed. Also returns whether the initial speed allows braking early enough for curves and the stop, and to not run
104// into the leader. `leader` holds the distance to the leader's rear and the leader's speed, at the start and at each
105// sample time.
106std::pair<bool, std::vector<std::pair<double, double>>> limitedMotionAlongRoute(
107 const lanelet::routing::LaneletPath& route,
108 double start_arc_length,
109 double speed,
110 double sample_interval,
111 std::size_t sample_count,
112 double max_lateral_acceleration,
113 double max_acceleration,
114 double max_deceleration,
115 double stop_distance,
116 const std::vector<std::pair<double, double>>& leader,
117 double min_distance) {
118 // Speed limit every meter ahead: the curve speed or zero beyond the stop, lowered so that the limits further ahead
119 // can be reached by braking
120 const auto limit_count = static_cast<std::size_t>(speed * sample_interval * static_cast<double>(sample_count)) + 2;
121 const std::vector<double> curvatures = curvaturesAlongLine(
122 lanelet::LaneletSequence(lanelet::ConstLanelets(route.begin(), route.end())).centerline2d(), start_arc_length, limit_count);
123 std::vector<double> speed_limits(limit_count, speed);
124 for (std::size_t i = limit_count; i-- > 0;) {
125 speed_limits[i] =
126 static_cast<double>(i) >= stop_distance ? 0.0 : std::min(speed, std::sqrt(max_lateral_acceleration / curvatures[i]));
127 if (i + 1 < limit_count) {
128 speed_limits[i] = std::min(speed_limits[i], std::sqrt(std::pow(speed_limits[i + 1], 2) + 2.0 * max_deceleration));
129 }
130 }
131
132 bool feasible = speed <= speed_limits.front();
133
134 // Simulate in small time steps, braking or accelerating towards the next speed limit. Behind a leader, the speed is
135 // also limited so that braking reaches the leader's speed `min_distance` behind it, assuming it keeps its speed.
136 const double dt = sample_interval / kSimulationStepsPerSample;
137 double distance = 0.0;
138 double previous_gap = std::numeric_limits<double>::infinity(); // infinite without a leader
139 std::vector<std::pair<double, double>> motion;
140 for (std::size_t step = 1; step <= sample_count * kSimulationStepsPerSample; ++step) {
141 const auto next_limit = std::min(static_cast<std::size_t>(std::ceil(distance)), limit_count - 1);
142 double target_speed = speed_limits[next_limit];
143 if (!leader.empty()) {
144 // Gap to the leader's rear now: its distance at the last sample, moved on at its speed since then
145 const std::size_t sample = (step - 1) / kSimulationStepsPerSample;
146 const auto [leader_distance, leader_speed] = leader[sample];
147 const double time_since_sample = dt * static_cast<double>(step - 1 - sample * kSimulationStepsPerSample);
148 const double gap = leader_distance + leader_speed * time_since_sample - distance;
149
150 // Highest speed from which braking still reaches the leader's speed `min_distance` behind it:
151 // v^2 = v_leader^2 + 2*a*(gap - min_distance)
152 if (gap > 0.0) {
153 const double gap_after_step = gap - (speed - leader_speed) * dt;
154 target_speed = std::min(
155 target_speed,
156 std::sqrt(std::max(0.0, std::pow(leader_speed, 2) + 2.0 * max_deceleration * (gap_after_step - min_distance))));
157 }
158
159 // Reaching the rear of the leader ahead in the previous step could not be avoided by braking
160 if (gap <= 0.0 && previous_gap > 0.0 && std::isfinite(previous_gap)) feasible = false;
161 previous_gap = gap;
162 }
163
164 // Approach the target speed within the acceleration limits
165 speed = std::clamp(target_speed, speed - max_deceleration * dt, speed + max_acceleration * dt);
166 distance += speed * dt;
167 if (step % kSimulationStepsPerSample == 0) motion.emplace_back(distance, speed);
168 }
169 return {feasible, motion};
170}

174// Distance from the object's front at `front_arc_length` along `route` to where it stops to yield, or infinity. The
175// yield rules of the route's lanelets and of the lanelets before it apply, as a rule is usually set on the last lanelet
176// before a junction, while its stop line can lie on a following lanelet. Without a stop line, the object stops at the
177// end of the regulated lanelet. It yields if another route or ego's route passes a priority lanelet, i.e. a right-of-way
178// lanelet of the rule or a lanelet following it, and then overlaps a non-priority lanelet of the route beyond the stop.
179double yieldStopDistance(const lanelet::routing::LaneletPath& route,
180 double front_arc_length,
181 const lanelet::routing::RoutingGraph& routing_graph,
182 const std::vector<lanelet::routing::LaneletPath>& priority_routes,
183 const std::vector<lanelet::BasicPoint2d>& ego_route) {
184 const lanelet::ConstLanelets previous = routing_graph.previous(route.front());
185 const lanelet::CompoundLineString2d centerline =
186 lanelet::LaneletSequence(lanelet::ConstLanelets(route.begin(), route.end())).centerline2d();
187 std::vector<double> lanelet_ends; // arc length of each route lanelet's end
188 for (const lanelet::ConstLanelet& lanelet : route) {
189 lanelet_ends.push_back((lanelet_ends.empty() ? 0.0 : lanelet_ends.back()) +
190 static_cast<double>(lanelet::geometry::length(lanelet.centerline2d())));
191 }
192
193 double stop_distance = std::numeric_limits<double>::infinity();
194 lanelet::ConstLanelets regulated = previous;
195 regulated.insert(regulated.end(), route.begin(), route.end());
196 for (std::size_t i = 0; i < regulated.size(); ++i) {
197 for (const auto& rule : regulated[i].regulatoryElementsAs<lanelet::RightOfWay>()) {
198 if (rule->getManeuver(regulated[i]) != lanelet::ManeuverType::Yield) continue;
199 double stop_arc_length = std::numeric_limits<double>::infinity();
200 if (const auto stop_line = rule->stopLine()) {
201 lanelet::BasicPoint2d center(0.0, 0.0);
202 for (const auto& point : *stop_line) center += point.basicPoint2d() / static_cast<double>(stop_line->size());
203 const lanelet::ArcCoordinates arc = lanelet::geometry::toArcCoordinates(centerline, center);
204 if (std::abs(arc.distance) <= kStopLineMaxOffsetM) stop_arc_length = arc.length;
205 } else if (i >= previous.size()) {
206 stop_arc_length = lanelet_ends[i - previous.size()];
207 }
208 if (stop_arc_length < front_arc_length || stop_arc_length - front_arc_length >= stop_distance) continue;
209
210 // Priority lanelets, and the conflict area: the non-priority route lanelets beyond the stop
211 lanelet::ConstLanelets priority = rule->rightOfWayLanelets();
212 for (const lanelet::ConstLanelet& lanelet : rule->rightOfWayLanelets()) {
213 const lanelet::ConstLanelets following = routing_graph.following(lanelet, false);
214 priority.insert(priority.end(), following.begin(), following.end());
215 }
216 const auto is_priority = [&](const lanelet::ConstLanelet& lanelet) {
217 return std::find(priority.begin(), priority.end(), lanelet) != priority.end();
218 };
219 lanelet::ConstLanelets conflict_area;
220 for (std::size_t j = 0; j < route.size(); ++j) {
221 if (lanelet_ends[j] > stop_arc_length && !is_priority(route[j])) conflict_area.push_back(route[j]);
222 }
223 const auto in = [](const lanelet::ConstLanelets& lanelets, const lanelet::BasicPoint2d& point) {
224 return std::any_of(lanelets.begin(), lanelets.end(), [&](const auto& l) { return lanelet::geometry::inside(l, point); });
225 };
226 const auto overlaps_conflict_area = [&](const lanelet::ConstLanelet& lanelet) {
227 return std::any_of(conflict_area.begin(), conflict_area.end(),
228 [&](const auto& c) { return lanelet::geometry::overlaps2d(lanelet, c); });
229 };
230
231 // Priority traffic first passes a priority lanelet and then overlaps the conflict area
232 bool conflict = std::any_of(priority_routes.begin(), priority_routes.end(), [&](const auto& other) {
233 return std::any_of(std::find_if(other.begin(), other.end(), is_priority), other.end(), overlaps_conflict_area);
234 });
235 const auto ego_entry = std::find_if(ego_route.begin(), ego_route.end(), [&](const auto& p) { return in(priority, p); });
236 conflict = conflict || std::any_of(ego_entry, ego_route.end(), [&](const auto& p) { return in(conflict_area, p); });
237 if (conflict) stop_distance = stop_arc_length - front_arc_length;
238 }
239 }
240 return stop_distance;
241}

243// Distance along `route` from the follower's front at `front_arc_length` to the rear of the closest leader, and that
244// leader's speed, at the start and at each sample time; infinite distance while there is no leader. A leader is on the
245// route while it is inside the route's lanelets and its yaw differs from the centerline's by at most `max_yaw_diff`, as
246// for matching objects to lanelets, so traffic crossing or against the route is not followed. It must start ahead of
247// the follower's front. Returns an empty vector if no leader is ever on the route.
248std::vector<std::pair<double, double>> closestLeaderAlongRoute(const lanelet::routing::LaneletPath& route,
249 double front_arc_length,
250 const std::vector<Leader>& leaders,
251 const perception_msgs::msg::Object* follower,
252 std::size_t sample_count,
253 double max_yaw_diff) {
254 if (leaders.empty()) return {};
255 const lanelet::LaneletSequence sequence(lanelet::ConstLanelets(route.begin(), route.end()));
256 const lanelet::BasicPolygon2d area = sequence.polygon2d().basicPolygon();
257 const lanelet::BoundingBox2d box = lanelet::geometry::boundingBox2d(area); // to quickly skip positions far away
258 const lanelet::BasicLineString2d centerline = sequence.centerline2d().basicLineString();
259
260 std::vector<std::pair<double, double>> closest(sample_count + 1, {std::numeric_limits<double>::infinity(), 0.0});
261 bool any_leader = false;
262 for (const Leader& leader : leaders) {
263 if (leader.object == follower) continue; // an object does not follow its own predictions
264 for (std::size_t sample = 0; sample < leader.positions.size() && sample < closest.size(); ++sample) {
265 // Distance along the route to the leader's rear, if the leader is on the route and heads along it
266 const lanelet::BasicPoint2d& position = leader.positions[sample];
267 std::optional<double> distance;
268 if (box.contains(position) && boost::geometry::covered_by(position, area)) {
269 const auto [segment_start, segment_end] = lanelet::geometry::closestSegment(centerline, position);
270 const lanelet::BasicPoint2d direction = segment_end - segment_start;
271 if (std::abs(wrap_angle_rad(leader.yaws[sample] - std::atan2(direction.y(), direction.x()))) <= max_yaw_diff) {
272 distance = lanelet::geometry::toArcCoordinates(centerline, position).length + leader.rear_offset - front_arc_length;
273 }
274 }
275
276 // A leader that starts behind or beside the follower is never followed, and positions behind the follower's
277 // start are skipped. Otherwise, the leader counts if it is the closest one at this sample time.
278 if (distance && *distance <= 0.0 && sample == 0) break;
279 if (!distance || *distance <= 0.0 || *distance >= closest[sample].first) continue;
280 closest[sample] = {*distance, leader.speeds[sample]};
281 any_leader = true;
282 }
283 }
284 return any_leader ? closest : std::vector<std::pair<double, double>>{};
285}

304// Share of a objects area covered by a lanelet and the lanelets connected to it ahead and behindlocaloc, so that
305// a footprint across lanelet joints counts fully, while neighboring lanes stay separate
306double laneSequenceCoverage(const BoostPolygon& footprint,
307 const lanelet::ConstLanelet& lanelet,
308 const lanelet::routing::RoutingGraph& graph) {
309 const auto overlap_with = [&footprint](const lanelet::ConstLanelet& other) {
310 BoostPolygon area;
311 for (const auto& point : other.polygon2d().basicPolygon()) area.outer().push_back(point);
312 boost::geometry::correct(area);
313 BoostMultiPolygon overlap;
314 boost::geometry::intersection(footprint, area, overlap);
315 return overlap;
316 };
317 BoostMultiPolygon covered = overlap_with(lanelet);
318 for (const bool ahead : {true, false}) {
319 std::vector<lanelet::ConstLanelet> open = {lanelet};
320 std::set<lanelet::Id> visited = {lanelet.id()};
321 while (!open.empty()) {
322 const lanelet::ConstLanelet current = open.back();
323 open.pop_back();
324 for (const auto& next : ahead ? graph.following(current, false) : graph.previous(current, false)) {
325 if (!visited.insert(next.id()).second) continue;
326 const BoostMultiPolygon overlap = overlap_with(next);
327 if (boost::geometry::area(overlap) <= 0.0) continue; // the sequence ends where it leaves the footprint
328 BoostMultiPolygon merged;
329 boost::geometry::union_(covered, overlap, merged);
330 covered = merged;
331 open.push_back(next);
332 }
333 }
334 }
335 return boost::geometry::area(covered) / boost::geometry::area(footprint);
336}

523void Lanelet2ObjectListPrediction::objectListCallback(const perception_msgs::msg::ObjectList::ConstSharedPtr& msg) {
524 RCLCPP_DEBUG(this->get_logger(), "Message received with stamp: '%d'", msg->header.stamp.sec);
525
526 if (!checkMap(true)) {
527 RCLCPP_WARN(this->get_logger(), "Lanelet2 map is not loaded yet, skipping object list");
528 return;
529 }
530
531 perception_msgs::msg::ObjectList object_list_map_frame;
532 if (msg->header.frame_id != ll2_interface_->map_frame_id_) {
533 try {
534 object_list_map_frame = tf_buffer_->transform(*msg, ll2_interface_->map_frame_id_, tf2::durationFromSec(0.1));
535 } catch (tf2::TransformException& ex) {
536 RCLCPP_ERROR(this->get_logger(), "Could not transform object list from frame '%s' to frame '%s': %s. Skipping object list.",
537 msg->header.frame_id.c_str(), ll2_interface_->map_frame_id_.c_str(), ex.what());
538 return;
539 }
540 } else {
541 object_list_map_frame = *msg;
542 }
543
544 // Routes of all objects first, so that yielding is evaluated against the nominal routes of the whole scene
545 std::vector<PredictionObject> prediction_objects = matchObjectListToMap(object_list_map_frame);
546 for (PredictionObject& prediction_object : prediction_objects) {
547 prediction_object.routes = createRoutes(prediction_object);
548 }
549
550 // Remaining route of ego in map frame along its suggested lanes
551 std::vector<lanelet::BasicPoint2d> ego_route;
552 if (yielding_enable_ && latest_route_ != nullptr && latest_route_->header.frame_id == ll2_interface_->map_frame_id_) {
553 const auto& elements = latest_route_->route_elements;
554 for (std::size_t i = latest_route_->current_route_element_idx; i < elements.size(); ++i) {
555 if (elements[i].suggested_lane_idx >= elements[i].lane_elements.size()) continue;
556 const geometry_msgs::msg::Point& position =
557 elements[i].lane_elements[elements[i].suggested_lane_idx].reference_pose.position;
558 ego_route.emplace_back(position.x, position.y);
559 }
560 }
561
562 // First pass: nominal predictions of all objects, including yielding, but without leaders
563 const builtin_interfaces::msg::Time& stamp = object_list_map_frame.header.stamp;
564 std::size_t matched_object_count = 0;
565 for (PredictionObject& prediction_object : prediction_objects) {
566 if (!prediction_object.lanelet_matches.empty()) {
567 ++matched_object_count;
568 }
569 prediction_object.object.state_predictions =
570 createPredictionsForMatchedObject(prediction_object, prediction_objects, ego_route, {}, stamp);
571 }
572
573 // Second pass: predictions again, slowing down behind the nominal predictions of the first pass and behind ego's
574 // planned trajectory. The leaders are copied before, so slowing down does not propagate to further followers.
575 if (following_enable_) {
576 const std::vector<Leader> leaders = createLeaders(prediction_objects, stamp);
577 for (PredictionObject& prediction_object : prediction_objects) {
578 prediction_object.object.state_predictions =
579 createPredictionsForMatchedObject(prediction_object, prediction_objects, ego_route, leaders, stamp);
580 }
581 }
582 RCLCPP_DEBUG(this->get_logger(), "Matched %zu/%zu objects to at least one lanelet", matched_object_count,
583 object_list_map_frame.objects.size());
584
585 perception_msgs::msg::ObjectList out_msg = object_list_map_frame;
586 out_msg.objects.clear();
587 out_msg.objects.reserve(prediction_objects.size());
588 for (const PredictionObject& prediction_object : prediction_objects) {
589 out_msg.objects.push_back(prediction_object.object);
590 }
591
592 publisher_->publish(out_msg);
593 RCLCPP_DEBUG(this->get_logger(), "Message published with stamp: '%d'", out_msg.header.stamp.sec);
594}

596std::vector<Lanelet2ObjectListPrediction::PredictionObject> Lanelet2ObjectListPrediction::matchObjectListToMap(
597 const perception_msgs::msg::ObjectList& object_list) const {
598 std::vector<PredictionObject> prediction_objects;
599 prediction_objects.reserve(object_list.objects.size());
600
601 const auto map = ll2_interface_->getMapPtr();
602 if (map == nullptr) {
603 RCLCPP_ERROR(this->get_logger(), "Lanelet2 map pointer is null, cannot match objects to lanelets");
604 return prediction_objects;
605 }
606
607 for (std::size_t object_index = 0; object_index < object_list.objects.size(); ++object_index) {
608 PredictionObject prediction_object;
609 prediction_object.object = object_list.objects[object_index];
610
611 using perception_msgs::msg::ObjectClassification;
612 const uint8_t type = perception_msgs::object_access::getClassWithHighestProbability(prediction_object.object).type;
613 const bool participant_specific = participant_specific_matching_enable_;
614 const bool pedestrian =
615 participant_specific && (type == ObjectClassification::PEDESTRIAN || type == ObjectClassification::VRU);
616 const bool two_wheeler =
617 participant_specific && (type == ObjectClassification::BICYCLE || type == ObjectClassification::MICRO ||
618 type == ObjectClassification::MOTORCYCLE);
619 // Without participant-specific matching, pedestrians are not matched, as they are not constrained to the road network
620 if (!participant_specific && type == ObjectClassification::PEDESTRIAN) {
621 prediction_objects.push_back(prediction_object);
622 continue;
623 }
624
625 geometry_msgs::msg::Point position;
626 double object_yaw = 0.0;
627 try {
628 position = perception_msgs::object_access::getPosition(prediction_object.object);
629 object_yaw = perception_msgs::object_access::getYaw(prediction_object.object);
630 } catch (const std::exception& ex) {
631 RCLCPP_WARN(this->get_logger(), "Could not read position or yaw of object %zu: %s", object_index, ex.what());
632 prediction_objects.push_back(prediction_object);
633 continue;
634 }
635
636 const lanelet::BasicPoint2d position_2d(position.x, position.y);
637
638 // Footprint of the object along the tracker's yaw
639 BoostPolygon footprint;
640 double search_distance = lanelet_match_max_distance_m_;
641 if (lane_coverage_matching_enable_) {
642 try {
643 const double length = perception_msgs::object_access::getLength(prediction_object.object);
644 const double width = perception_msgs::object_access::getWidth(prediction_object.object);
645 const double yaw = perception_msgs::object_access::getYaw(prediction_object.object);
646 const auto& offset = prediction_object.object.state.reference_point.translation_to_geometric_center;
647 const lanelet::BasicPoint2d forward(std::cos(yaw), std::sin(yaw));
648 const lanelet::BasicPoint2d left(-std::sin(yaw), std::cos(yaw));
649 const lanelet::BasicPoint2d center = position_2d + offset.x * forward + offset.y * left;
650 for (const auto& [f, l] : {std::pair{0.5, 0.5}, {0.5, -0.5}, {-0.5, -0.5}, {-0.5, 0.5}}) {
651 footprint.outer().push_back(center + f * length * forward + l * width * left);
652 }
653 boost::geometry::correct(footprint);
654 if (boost::geometry::area(footprint) > 0.0) {
655 search_distance = 0.5 * std::hypot(length, width) + std::hypot(offset.x, offset.y);
656 } else {
657 footprint.clear();
658 }
659 } catch (const std::exception& ex) {
660 RCLCPP_DEBUG(this->get_logger(), "Could not read the footprint of object %zu: %s", object_index, ex.what());
661 }
662 }
663 const auto candidate_lanelets = lanelet::geometry::findWithin2d(map->laneletLayer, position_2d, search_distance);
664
665 for (const auto& candidate_lanelet : candidate_lanelets) {
666 lanelet::ConstLanelet lanelet = candidate_lanelet.second;
667 lanelet::ConstLanelet matched_lanelet = lanelet;
668 double start_arc_length = lanelet::geometry::toArcCoordinates(lanelet.centerline2d(), position_2d).length;
669 const double lanelet_length = static_cast<double>(lanelet::geometry::length(lanelet.centerline2d()));
670 start_arc_length = std::clamp(start_arc_length, 0.0, lanelet_length);
671 double orientation_difference = 0.0;
672
673 const double lanelet_yaw = computeLaneletYawAtArcLength(lanelet, start_arc_length);
674 const double inverted_arc_length = lanelet_length - start_arc_length;
675 const double inverted_yaw = computeLaneletYawAtArcLength(lanelet.invert(), inverted_arc_length);
676 const double lanelet_difference = std::abs(wrap_angle_rad(object_yaw - lanelet_yaw));
677 const double inverted_difference = std::abs(wrap_angle_rad(object_yaw - inverted_yaw));
678
679 if (inverted_difference < lanelet_difference) {
680 matched_lanelet = lanelet.invert();
681 start_arc_length = inverted_arc_length;
682 orientation_difference = inverted_difference;
683 } else {
684 orientation_difference = lanelet_difference;
685 }
686
687 if (orientation_difference > lanelet_match_max_yaw_diff_rad_) {
688 continue;
689 }
690
691 // Pedestrians and bicycles follow their own traffic rules, motorcycles only on bicycle lanes
692 const ParticipantRouting* routing = &vehicle_routing_;
693 if (pedestrian) {
694 routing = &pedestrian_routing_;
695 } else if (two_wheeler && (type != ObjectClassification::MOTORCYCLE || isBicycleLane(matched_lanelet))) {
696 routing = &bicycle_routing_;
697 }
698 if (routing->traffic_rules == nullptr) {
699 continue;
700 }
701 const bool opposite_direction = !routing->traffic_rules->canPass(matched_lanelet);
702 if (opposite_direction && !(two_wheeler && participant_specific_matching_allow_opposite_direction_ &&
703 routing->traffic_rules->canPass(matched_lanelet.invert()))) {
704 continue;
705 }
706
707 prediction_object.lanelet_matches.push_back(LaneletMatch{matched_lanelet, candidate_lanelet.first, start_arc_length,
708 orientation_difference, routing->routing_graph.get(),
709 opposite_direction});
710 }
711
712 // Two-wheelers prefer legal matches over matches against a lanelet's direction, then bicycle lanes over roads
713 auto& matches = prediction_object.lanelet_matches;
714 const auto rank = [](const LaneletMatch& match) {
715 return (match.opposite_direction ? 2 : 0) + (isBicycleLane(match.lanelet) ? 0 : 1);
716 };
717 if (two_wheeler && !matches.empty()) {
718 const int best_rank = rank(*std::min_element(
719 matches.begin(), matches.end(), [&](const LaneletMatch& a, const LaneletMatch& b) { return rank(a) < rank(b); }));
720 matches.erase(std::remove_if(matches.begin(), matches.end(), [&](const auto& match) { return rank(match) > best_rank; }),
721 matches.end());
722 }
723
724 // With footprint matching, the nearest matches count (several where lanelets overlap), if enough of
725 // the footprint lies on their lanelet sequence
726 if (lane_coverage_matching_enable_ && !matches.empty()) {
727 const double nearest = std::min_element(matches.begin(), matches.end(), [](const auto& a, const auto& b) {
728 return a.distance < b.distance;
729 })->distance;
730 matches.erase(std::remove_if(matches.begin(), matches.end(),
731 [&](const LaneletMatch& match) {
732 return match.distance > nearest ||
733 (!footprint.outer().empty() &&
734 laneSequenceCoverage(footprint, match.lanelet, *match.routing_graph) <
735 lane_coverage_matching_min_coverage_);
736 }),
737 matches.end());
738 }
739
740 if (prediction_object.lanelet_matches.empty()) {
741 RCLCPP_DEBUG(this->get_logger(), "Object %zu did not match any lanelet within %.2f m", object_index,
742 lanelet_match_max_distance_m_);
743 } else {
744 RCLCPP_DEBUG(this->get_logger(), "Object %zu matched to %zu lanelet candidate(s)", object_index,
745 prediction_object.lanelet_matches.size());
746 }
747 prediction_objects.push_back(prediction_object);
748 }
749
750 return prediction_objects;
751}

753std::vector<std::pair<Lanelet2ObjectListPrediction::LaneletMatch, lanelet::routing::LaneletPath>>
754Lanelet2ObjectListPrediction::createRoutes(const PredictionObject& prediction_object) const {
755 std::vector<std::pair<LaneletMatch, lanelet::routing::LaneletPath>> routes;
756 double speed = 0.0;
757 try {
758 speed = perception_msgs::object_access::getVelocityMagnitude(prediction_object.object);
759 } catch (const std::exception& ex) {
760 RCLCPP_WARN(this->get_logger(), "Could not read velocity for lanelet prediction: %s", ex.what());
761 return routes;
762 }
763 const double max_travel_distance = speed * prediction_sample_interval_s_ * static_cast<double>(getPredictionSampleCount());
764
765 for (const LaneletMatch& match : prediction_object.lanelet_matches) {
766 lanelet::routing::LaneletPaths match_routes;
767 if (max_travel_distance <= std::numeric_limits<double>::epsilon()) {
768 match_routes.push_back(lanelet::routing::LaneletPath({match.lanelet}));
769 } else if (match.opposite_direction) {
770 match_routes = oppositeDirectionPaths({match.lanelet}, max_travel_distance + match.start_arc_length, *match.routing_graph);
771 } else {
772 lanelet::routing::PossiblePathsParams params;
773 params.routingCostLimit = max_travel_distance + match.start_arc_length;
774 params.includeShorterPaths = true;
775 params.includeLaneChanges = false;
776 try {
777 match_routes = match.routing_graph->possiblePaths(match.lanelet, params);
778 } catch (const std::exception& ex) {
779 RCLCPP_WARN(this->get_logger(), "Could not create lanelet routes from matched lanelet: %s", ex.what());
780 continue;
781 }
782 if (match_routes.empty()) {
783 match_routes.push_back(lanelet::routing::LaneletPath({match.lanelet}));
784 }
785 }
786 for (const lanelet::routing::LaneletPath& route : match_routes) routes.emplace_back(match, route);
787 }
788 return routes;
789}

791std::vector<Leader> Lanelet2ObjectListPrediction::createLeaders(const std::vector<PredictionObject>& prediction_objects,
792 const builtin_interfaces::msg::Time& base_time) const {
793 std::vector<Leader> leaders;
794
795 // Appends the position, yaw and speed of a state as the leader's next sample
796 const auto add_sample = [](Leader& leader, const perception_msgs::msg::ObjectState& state) {
797 const geometry_msgs::msg::Point position = perception_msgs::object_access::getPosition(state);
798 leader.positions.emplace_back(position.x, position.y);
799 leader.yaws.push_back(perception_msgs::object_access::getYaw(state));
800 leader.speeds.push_back(perception_msgs::object_access::getVelocityMagnitude(state));
801 };
802
803 // Each prediction of an object is a leader, starting at the object's current state. The rear offset goes from the
804 // reference point to the rear, along the heading.
805 for (const PredictionObject& prediction_object : prediction_objects) {
806 const perception_msgs::msg::Object& object = prediction_object.object;
807 for (const perception_msgs::msg::ObjectStatePrediction& prediction : object.state_predictions) {
808 try {
809 const double length = perception_msgs::object_access::getLength(object);
810 Leader leader{&object, object.state.reference_point.translation_to_geometric_center.x - 0.5 * length, {}, {}, {}};
811 add_sample(leader, object.state);
812 for (const perception_msgs::msg::ObjectState& state : prediction.states) add_sample(leader, state);
813 leaders.push_back(leader);
814 } catch (const std::exception& ex) {
815 RCLCPP_DEBUG(this->get_logger(), "Could not read the state of a leading object: %s", ex.what());
816 }
817 }
818 }
819
820 // Ego at the first point of its planned trajectory at or after each prediction time, while the trajectory covers it
821 if (latest_ego_data_ == nullptr || latest_ego_data_->header.frame_id != ll2_interface_->map_frame_id_) return leaders;
822 const std::vector<perception_msgs::msg::ObjectState>& trajectory = latest_ego_data_->trajectory_planned;
823 const double ego_center = latest_ego_data_->state.reference_point.translation_to_geometric_center.x;
824 Leader ego{nullptr, ego_center - 0.5 * latest_ego_data_->length, {}, {}, {}};
825 try {
826 for (std::size_t sample = 0; sample <= getPredictionSampleCount(); ++sample) {
827 const rclcpp::Time time =
828 rclcpp::Time(base_time) + rclcpp::Duration::from_seconds(prediction_sample_interval_s_ * static_cast<double>(sample));
829 const auto point = std::find_if(trajectory.begin(), trajectory.end(),
830 [&](const auto& state) { return rclcpp::Time(state.header.stamp) >= time; });
831 if (point == trajectory.end()) break;
832 add_sample(ego, *point);
833 }
834 } catch (const std::exception& ex) {
835 RCLCPP_DEBUG(this->get_logger(), "Could not read ego's planned trajectory: %s", ex.what());
836 }
837 if (!ego.positions.empty()) leaders.push_back(ego);
838 return leaders;
839}

863std::vector<perception_msgs::msg::ObjectStatePrediction> Lanelet2ObjectListPrediction::createMapBasedPredictions(
864 const PredictionObject& prediction_object,
865 const std::vector<PredictionObject>& scene,
866 const std::vector<lanelet::BasicPoint2d>& ego_route,
867 const std::vector<Leader>& leaders,
868 const builtin_interfaces::msg::Time& base_time) const {
869 std::vector<perception_msgs::msg::ObjectStatePrediction> predictions;
870 double speed = 0.0;
871 double front_offset = 0.0; // from the object's position to its front
872 try {
873 speed = perception_msgs::object_access::getVelocityMagnitude(prediction_object.object);
874 front_offset = 0.5 * perception_msgs::object_access::getLength(prediction_object.object) +
875 prediction_object.object.state.reference_point.translation_to_geometric_center.x;
876 } catch (const std::exception& ex) {
877 RCLCPP_DEBUG(this->get_logger(), "Could not read velocity or length for lanelet prediction: %s", ex.what());
878 }
879 const std::size_t sample_count = getPredictionSampleCount();
880 const double max_travel_distance = speed * prediction_sample_interval_s_ * static_cast<double>(sample_count);
881 const geometry_msgs::msg::Point position = perception_msgs::object_access::getPosition(prediction_object.object);
882
883 // Routes of all other objects, which have right of way where they pass a right-of-way lanelet
884 std::vector<lanelet::routing::LaneletPath> priority_routes;
885 for (const PredictionObject& other : scene) {
886 if (!yielding_enable_ || &other == &prediction_object) continue;
887 for (const auto& route : other.routes) priority_routes.push_back(route.second);
888 }
889
890 for (const auto& [match, lanelet_route] : prediction_object.routes) {
891 // Distance to stop at a yield line for conflicting priority traffic
892 const double stop_distance = yielding_enable_ && !match.opposite_direction
893 ? yieldStopDistance(lanelet_route, match.start_arc_length + front_offset,
894 *match.routing_graph, priority_routes, ego_route)
895 : std::numeric_limits<double>::infinity();
896
897 // Closest leader ahead on the route
898 const std::vector<std::pair<double, double>> leader =
899 closestLeaderAlongRoute(lanelet_route, match.start_arc_length + front_offset, leaders, &prediction_object.object,
900 sample_count, lanelet_match_max_yaw_diff_rad_);
901
902 // Travel distance and speed at each sample
903 bool feasible = true;
904 std::vector<std::pair<double, double>> motion;
905 if ((motion_limits_enable_ || std::isfinite(stop_distance) || !leader.empty()) &&
906 max_travel_distance > std::numeric_limits<double>::epsilon()) {
907 std::tie(feasible, motion) = limitedMotionAlongRoute(
908 lanelet_route, match.start_arc_length, speed, prediction_sample_interval_s_, sample_count,
909 motion_limits_enable_ ? motion_limits_max_lateral_acceleration_mps2_ : std::numeric_limits<double>::infinity(),
910 max_longitudinal_acceleration_mps2_, max_longitudinal_deceleration_mps2_, stop_distance, leader,
911 following_min_distance_m_);
912 if (!feasible && infeasible_prediction_probability_ <= 0.0) continue; // not published without probability
913 } else {
914 for (std::size_t sample_index = 0; sample_index < sample_count; ++sample_index) {
915 motion.emplace_back(speed * prediction_sample_interval_s_ * static_cast<double>(sample_index + 1), speed);
916 }
917 }
918
919 // The object keeps its lateral position in the lane relative to the lane width, or follows the centerline
920 const double lane_offset =
921 keep_lane_offset_ ? relativeLaneOffset(match.lanelet, lanelet::BasicPoint2d(position.x, position.y)) : 0.0;
922
923 perception_msgs::msg::ObjectStatePrediction prediction;
924 prediction.states.reserve(sample_count);
925 for (std::size_t sample_index = 0; sample_index < sample_count; ++sample_index) {
926 prediction.states.push_back(sampleStateOnLaneletRoute(prediction_object.object.state, lanelet_route, match.start_arc_length,
927 motion[sample_index].first, motion[sample_index].second, lane_offset,
928 base_time, sample_index));
929 }
930 prediction.probability = feasible ? 1.0 : 0.0; // marks feasibility until the probabilities are assigned below
931 predictions.push_back(prediction);
932 }
933
934 // Infeasible predictions get the configured probability, feasible ones share the rest equally. Infeasible predictions
935 // without probability (e.g. <= 0.0) are not published. Without any feasible prediction, the object gets the fallback
936 // prediction instead.
937 const auto count = static_cast<double>(predictions.size());
938 const auto feasible_count = static_cast<double>(
939 std::count_if(predictions.begin(), predictions.end(), [](const auto& prediction) { return prediction.probability > 0.0; }));
940 if (feasible_count == 0.0) return {};
941 const double infeasible_probability = std::min(infeasible_prediction_probability_, 1.0 / count);
942 const double feasible_probability = (1.0 - (count - feasible_count) * infeasible_probability) / feasible_count;
943 for (perception_msgs::msg::ObjectStatePrediction& prediction : predictions) {
944 prediction.probability = prediction.probability > 0.0 ? feasible_probability : infeasible_probability;
945 }
946 return predictions;
947}

1000perception_msgs::msg::ObjectState Lanelet2ObjectListPrediction::sampleStateOnLaneletRoute(
1001 const perception_msgs::msg::ObjectState& base_state,
1002 const lanelet::routing::LaneletPath& route,
1003 double start_arc_length,
1004 double travel_distance,
1005 double speed,
1006 double lane_offset,
1007 const builtin_interfaces::msg::Time& base_time,
1008 std::size_t sample_index) const {
1009 perception_msgs::msg::ObjectState state = base_state;
1010 state.continuous_state_covariance.clear(); // explicitly don't model covariance for predicted states
1011 if (route.empty()) {
1012 RCLCPP_WARN(this->get_logger(), "Lanelet route is empty, cannot sample lanelet prediction");
1013 return state;
1014 }
1015
1016 double distance_on_route = start_arc_length + travel_distance;
1017 geometry_msgs::msg::Point fallback_position;
1018 try {
1019 fallback_position = perception_msgs::object_access::getPosition(base_state);
1020 } catch (const std::exception&) {
1021 fallback_position.x = 0.0;
1022 fallback_position.y = 0.0;
1023 fallback_position.z = 0.0;
1024 }
1025
1026 for (std::size_t route_index = 0; route_index < route.size(); ++route_index) {
1027 const lanelet::ConstLineString2d centerline = route[route_index].centerline2d();
1028 if (centerline.size() < 2) {
1029 continue;
1030 }
1031
1032 const double lanelet_length = static_cast<double>(lanelet::geometry::length(centerline));
1033 const bool is_last_lanelet = route_index + 1 == route.size();
1034 if (distance_on_route > lanelet_length && !is_last_lanelet) {
1035 distance_on_route -= lanelet_length;
1036 continue;
1037 }
1038
1039 const double arc_length = std::clamp(distance_on_route, 0.0, lanelet_length);
1040 lanelet::BasicPoint2d point = lanelet::geometry::interpolatedPointAtDistance(centerline, arc_length);
1041 const double yaw_sample_distance = std::min(0.5, std::max(0.01, lanelet_length * 0.1));
1042 double before_arc_length = std::max(0.0, arc_length - yaw_sample_distance);
1043 double after_arc_length = std::min(lanelet_length, arc_length + yaw_sample_distance);
1044 if (after_arc_length <= before_arc_length) {
1045 before_arc_length = 0.0;
1046 after_arc_length = lanelet_length;
1047 }
1048
1049 const lanelet::BasicPoint2d before_point = lanelet::geometry::interpolatedPointAtDistance(centerline, before_arc_length);
1050 const lanelet::BasicPoint2d after_point = lanelet::geometry::interpolatedPointAtDistance(centerline, after_arc_length);
1051 const double yaw = std::atan2(after_point.y() - before_point.y(), after_point.x() - before_point.x());
1052 if (lane_offset != 0.0) {
1053 const double half_width = laneHalfWidth(route[route_index], arc_length, lane_offset > 0.0);
1054 point += lane_offset * half_width * lanelet::BasicPoint2d(-std::sin(yaw), std::cos(yaw));
1055 }
1056 geometry_msgs::msg::Vector3 velocity;
1057 velocity.x = speed * std::cos(yaw);
1058 velocity.y = speed * std::sin(yaw);
1059 velocity.z = 0.0;
1060 setPredictedStateKinematics(state, point.x(), point.y(), fallback_position.z, yaw, velocity, base_time, sample_index);
1061 return state;
1062 }
1063
1064 const lanelet::ConstLineString2d last_centerline = route.back().centerline2d();
1065 const double last_length = static_cast<double>(lanelet::geometry::length(last_centerline));
1066 const lanelet::BasicPoint2d point = lanelet::geometry::interpolatedPointAtDistance(last_centerline, last_length);
1067 geometry_msgs::msg::Vector3 velocity;
1068 velocity.x = 0.0;
1069 velocity.y = 0.0;
1070 velocity.z = 0.0;
1071 setPredictedStateKinematics(state, point.x(), point.y(), fallback_position.z, 0.0, velocity, base_time, sample_index);
1072 return state;
1073}
